export interface HeadingProps extends React.HTMLAttributes<HTMLHeadingElement> {
  /**
   * The level of the heading
   * @default '1'
   */
  level?: 1 | 2 | 3 | 4 | 5 | 6;

  /**
   * The size variant of the heading
   * @default 'md' for h1, 'sm' for h2, 'xs' for h3, and 'text-xl' for the rest
   */
  size?: "2xl" | "xl" | "lg" | "md" | "sm" | "xs";

  /**
   * The font weight of the heading
   * @default 'bold'
   */
  weight?: "regular" | "medium" | "semibold" | "bold";

  /**
   * Whether the heading should be italic
   * @default false
   */
  italic?: boolean;

  /**
   * Whether the heading should be underlined
   * @default false
   */
  underline?: boolean;

  /**
   * The content to be rendered
   */
  children: React.ReactNode;
}

/**
 * Heading component that follows the design system guidelines
 */
export const Heading: React.FC<HeadingProps> = ({
  level = 1,
  size,
  weight = "bold",
  italic = false,
  underline = false,
  className = "",
  children,
  ...props
}) => {
  // Define possible size values explicitly
  type SizeKey = "2xl" | "xl" | "lg" | "md" | "sm" | "xs" | "text-xl" | "text-lg" | "text-md";

  // Default sizes based on heading level if not explicitly provided
  const defaultSizes: Record<number, SizeKey> = {
    1: "md",
    2: "sm",
    3: "xs",
    4: "text-xl",
    5: "text-lg",
    6: "text-md",
  };

  // Determine the final size, ensuring it's a valid SizeKey
  const finalSize: SizeKey = size || defaultSizes[level];

  const sizeClasses: Record<SizeKey, string> = {
    "2xl": "text-display-2xl",
    xl: "text-display-xl",
    lg: "text-display-lg",
    md: "text-display-md",
    sm: "text-display-sm",
    xs: "text-display-xs",
    "text-xl": "text-text-xl", // Renamed from text-xl to avoid conflict with Tailwind
    "text-lg": "text-text-lg",   // Renamed from text-lg
    "text-md": "text-text-md",   // Renamed from text-md
  };

  const weightClasses = {
    regular: "font-regular",
    medium: "font-medium",
    semibold: "font-semibold",
    bold: "font-bold",
  };

  const italicClass = italic ? "italic" : "";
  const underlineClass = underline ? "underline" : "";

  const classes = cn(
    sizeClasses[finalSize],
    weightClasses[weight],
    italicClass,
    underlineClass,
    className
  );

  // HeadingTag is now just a string like "h1", "h2", etc.
  const HeadingTag = `h${level}`;

  return React.createElement(
    HeadingTag, // Use the string tag name directly
    { className: classes, ...props },
    children
  );
};

export type ThemeColor = 'brand' | 'error' | 'warning' | 'success';
export type ComponentVariant = 'primary' | 'secondary' | 'tertiary';

// Map design system colors to shadcn variants
export const mapColorToShadcnVariant = (color: ThemeColor) => {
  switch (color) {
    case 'brand': return 'default';
    case 'error': return 'destructive';
    case 'warning':
    case 'success':
      return 'secondary'; // Will need additional classes
    default: return 'default';
  }
};

// Map design system variants to shadcn variants
export const mapVariantToShadcnVariant = (variant: ComponentVariant) => {
  switch (variant) {
    case 'primary': return 'default';
    case 'secondary': return 'outline';
    case 'tertiary': return 'ghost';
    default: return 'default';
  }
};

// Generate tailwind classes for each color/variant combination

  color: ThemeColor,

// src/components/ui/button/button.tsx
import * as React from "react";
import { cn } from "@/lib/utils";
import { ThemeColor, ComponentVariant, getColorVariantClasses } from "@/lib/theme-utils";


export interface ButtonProps extends React.ButtonHTMLAttributes<HTMLButtonElement> {
  /**
   * The visual style of the button
   * @default 'primary'
   */
  variant?: ComponentVariant;

  /**
   * The color theme of the button
   * @default 'brand'
   */
  color?: ThemeColor;

  /**
   * The size of the button
   * @default 'md'
   */
  size?: "sm" | "md" | "lg";

  /**
   * Optional icon to display before the button text
   */
  leftIcon?: React.ReactNode;

  /**
   * Optional icon to display after the button text
   */
  rightIcon?: React.ReactNode;

  /**
   * Whether the button is in a loading state
   * @default false
   */
  isLoading?: boolean;
}

/**
 * Button component that follows the design system guidelines
 */
export const Button = React.forwardRef<HTMLButtonElement, ButtonProps>(
  ({ 
    className, 
    variant = "primary", 
    size = "md", 
    color = "brand",
    isLoading = false, 
    leftIcon, 
    rightIcon,
    disabled,
    children, 
    ...props 
  }, ref) => {
    // Get base size classes
    const sizeClasses = {
      sm: "px-3 py-1.5 text-sm h-9",
      md: "px-4 py-2 text-md h-10",
      lg: "px-5 py-2.5 text-lg h-11",
    };
    
    // Get icon size based on button size
    const iconSize = {
      sm: 16,
      md: 18, 
      lg: 20
    };
    
    // Get color+variant specific classes
    const colorClasses = getColorVariantClasses(color, variant);
    
    // Combine all classes
    const buttonClasses = cn(
      "inline-flex items-center justify-center rounded-md font-medium transition-colors",
      "focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-offset-2",
      "disabled:opacity-50 disabled:pointer-events-none",
      sizeClasses[size],
      colorClasses,
      className
    );
    
    return (
      <button
        ref={ref}
        className={buttonClasses}
        disabled={disabled || isLoading}
        type="button"
        {...props}
      >
        {leftIcon && <span className="mr-2">{leftIcon}</span>}
        {isLoading ? (
          <span className="flex items-center">
            <svg className="animate-spin -ml-1 mr-2 h-4 w-4 text-current" xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24">
              <circle className="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" strokeWidth="4"></circle>
              <path className="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"></path>
            </svg>
            Loading...
          </span>
        ) : (
          children
        )}
        {rightIcon && <span className="ml-2">{rightIcon}</span>}
      </button>
    );
  }
);

Button.displayName = "Button";

// src/lib/theme-utils.ts
export type ThemeColor = 'brand' | 'error' | 'warning' | 'success';
export type ComponentVariant = 'primary' | 'secondary' | 'tertiary';
import { clsx, type ClassValue } from 'clsx';
import { twMerge } from 'tailwind-merge';

// Map design system colors to shadcn variants
export const mapColorToShadcnVariant = (color: ThemeColor) => {
  switch (color) {
    case 'brand': return 'default';
    case 'error': return 'destructive';
    case 'warning':
    case 'success':
      return 'secondary'; // Will need additional classes
    default: return 'default';
  }
};

// Map design system variants to shadcn variants
export const mapVariantToShadcnVariant = (variant: ComponentVariant) => {
  switch (variant) {
    case 'primary': return 'default';
    case 'secondary': return 'outline';
    case 'tertiary': return 'ghost';
    default: return 'default';
  }
};

//Combines multiple class values into a single className string with Tailwind-aware merging
export function cn(...inputs: ClassValue[]) {
  return twMerge(clsx(inputs));
}

// Generate tailwind classes for each color/variant combination
export const getColorVariantClasses = (
  color: ThemeColor, 
  variant: ComponentVariant
): string => {
  const colorVariantMap = {
    primary: {
      brand: 'bg-brand-600 hover:bg-brand-700 text-white focus-visible:ring-brand-500',
      error: 'bg-error-600 hover:bg-error-700 text-white focus-visible:ring-error-500',
      warning: 'bg-warning-600 hover:bg-warning-700 text-white focus-visible:ring-warning-500',
      success: 'bg-success-600 hover:bg-success-700 text-white focus-visible:ring-success-500',
    },
    secondary: {
      brand: 'border-brand-300 text-brand-700 hover:bg-brand-50 focus-visible:ring-brand-500',
      error: 'border-error-300 text-error-700 hover:bg-error-50 focus-visible:ring-error-500',
      warning: 'border-warning-300 text-warning-700 hover:bg-warning-50 focus-visible:ring-warning-500',
      success: 'border-success-300 text-success-700 hover:bg-success-50 focus-visible:ring-success-500',
    },
    tertiary: {
      brand: 'text-brand-700 hover:bg-brand-50 focus-visible:ring-brand-500',
      error: 'text-error-700 hover:bg-error-50 focus-visible:ring-error-500',
      warning: 'text-warning-700 hover:bg-warning-50 focus-visible:ring-warning-500',
      success: 'text-success-700 hover:bg-success-50 focus-visible:ring-success-500',
    }
  };

  return colorVariantMap[variant]?.[color] || '';
};
