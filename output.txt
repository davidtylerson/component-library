/**
 * Generic types to support polymorphic components
 */

// Base type for props without ref
export type AsProp<C extends React.ElementType> = {
    /**
     * An override of the default HTML tag.
     * @default based on component default
     */
    as?: C;
  };
  
  // Props that should be excluded from component props when defining "as" prop
  export type PropsToOmit<C extends React.ElementType, P> = keyof (AsProp<C> & P);
  
  // Polymorphic component props (without ref)
  export type PolymorphicComponentProps<
    C extends React.ElementType,
    Props = {}
  > = React.PropsWithChildren<Props & AsProp<C>> &
    Omit<React.ComponentPropsWithoutRef<C>, PropsToOmit<C, Props>>;
  
  // Polymorphic component props (with ref)
  export type PolymorphicComponentPropsWithRef<
    C extends React.ElementType,
    Props = {}
  > = PolymorphicComponentProps<C, Props> & { ref?: PolymorphicRef<C> };
  
  // For polymorphic components that need refs
  export type PolymorphicRef<C extends React.ElementType> =
    React.ComponentPropsWithRef<C>["ref"];
  
  // For creating polymorphic components with proper ref forwarding
  export type PolymorphicComponentWithRef<
    Props,
    DefaultElementType extends React.ElementType
  > = <C extends React.ElementType = DefaultElementType>(
    props: PolymorphicComponentPropsWithRef<C, Props>
  ) => React.ReactElement | null;

import React from 'react';
import './button.css';

interface ButtonProps {
  /**
   * Is this the principal call to action on the page?
   */
  primary?: boolean;
  /**
   * What background color to use
   */
  backgroundColor?: string;
  /**
   * How large should the button be?
   */
  size?: 'small' | 'medium' | 'large';
  /**
   * Button contents
   */
  label: string;
  /**
   * Optional click handler
   */
  onClick?: () => void;
}

/**
 * Primary UI component for user interaction
 */
export const Button = ({
  primary = false,
  size = 'medium',
  backgroundColor,
  label,
  ...props
}: ButtonProps) => {
  const mode = primary ? 'storybook-button--primary' : 'storybook-button--secondary';
  return (
    <button
      type="button"
      className={['storybook-button', `storybook-button--${size}`, mode].join(' ')}
      {...props}
    >
      {label}
      <style jsx>{`
        button {
          background-color: ${backgroundColor};
        }
      `}</style>
    </button>
  );
};

/**
 * Heading component that follows the design system guidelines.
 * 
 * This component renders headings following the design system's typography

 */
export const Heading = React.forwardRef<HTMLHeadingElement, HeadingProps>(
  ({ 
    level = 1,
    size,
    weight = "bold",
    italic = false,
    underline = false,
    color = "default",
    className = "",
    as,
    children,
    ...props
  }, ref) => {
    // Default heading tag based on level
    const defaultTag = `h${level}` as ElementType;
    
    // Use provided "as" prop or default based on level
    const Component = as || defaultTag;
    
    // Default sizes based on heading level if not explicitly provided

      1: "md",
      2: "sm",
      3: "xs",
      4: "text-xl",
      5: "text-lg",
      6: "text-md",
    };

    // Determine the final size
    const finalSize = size || defaultSizes[level];

    const sizeClasses: Record<string, string> = {

      "xl": "text-display-xl",
      "lg": "text-display-lg",
      "md": "text-display-md",

"use client"

import React, { createContext, useContext, useEffect, useState } from "react"
import { useTheme as useNextTheme } from "next-themes"
import { isDarkTheme } from "@/lib/theme-utils"

/**
 * Theme context type definition
 */
type ThemeContextType = {
  /** Current theme name (light, dark, system) */
  theme: string;
  
  /** Method to change the theme */
  setTheme: (theme: string) => void;
  
  /** Current resolved theme (light or dark) accounting for system preference */
  resolvedTheme: string;
  
  /** Whether the client-side code has been initialized */
  isClient: boolean;
  
  /** Whether the current theme is dark */
  isDark: boolean;
}

const ThemeContext = createContext<ThemeContextType | undefined>(undefined);

/**
 * ThemeContextProvider wraps the app and provides theme context
 */
export function ThemeContextProvider({ 
  children 
}: { 
  children: React.ReactNode 
}) {
  const [isClient, setIsClient] = useState(false);
  const { theme, setTheme, resolvedTheme, systemTheme } = useNextTheme();

  // Handle client-side hydration
  useEffect(() => {
    setIsClient(true);
  }, []);

  // Determine if the theme is dark
  const isDark = isDarkTheme(theme, systemTheme);

  return (
    <ThemeContext.Provider value={{
      theme: theme || "system",
      setTheme,
      resolvedTheme: resolvedTheme || "light",
      isClient,
      isDark,
    }}>
      {children}
    </ThemeContext.Provider>
  );
}

/**
 * Custom hook to access theme context
 * 
 * @returns Theme context object
 */
export function useTheme() {
  const context = useContext(ThemeContext);
  
  if (context === undefined) {
    throw new Error("useTheme must be used within a ThemeContextProvider");
  }
  
  return context;
}

/**
 * Props for the ThemeToggle component
 */
export interface ThemeToggleProps extends React.HTMLAttributes<HTMLButtonElement> {
  /**
   * Variant of the toggle
   * @default 'icon'
   */
  variant?: 'icon' | 'switch' | 'button';
  
  /**
   * Size of the toggle
   * @default 'md'
   */
  size?: 'sm' | 'md' | 'lg';
}

/**
 * ThemeToggle component to switch between light and dark themes
 */
export function ThemeToggle({ 
  className, 
  variant = 'icon', 
  size = 'md',
  ...props 
}: ThemeToggleProps) {
  const { theme, setTheme, resolvedTheme, isClient } = useTheme()

  // Only render on client to prevent hydration mismatch
  if (!isClient) return null;

  // Toggle between light and dark
  const toggleTheme = () => {
    setTheme(resolvedTheme === "dark" ? "light" : "dark")
  }

  // Size mapping for Tabler icons
  const iconSize = {
    sm: 16,
    md: 20,
    lg: 24,
  }

  // Size classes for button
  const sizeClasses = {
    sm: 'h-8 w-8',
    md: 'h-10 w-10',
    lg: 'h-12 w-12',
  }

  // Text size classes
  const textSizeClasses = {
    sm: 'text-xs',
    md: 'text-sm',
    lg: 'text-base',
  }

  // Variant specific rendering
  if (variant === 'icon') {
    return (
      <button
        type="button"
        onClick={toggleTheme}
        className={cn(
          "inline-flex items-center justify-center rounded-md border border-input bg-background hover:bg-accent hover:text-accent-foreground",
          sizeClasses[size],
          className
        )}
        aria-label={resolvedTheme === "dark" ? "Switch to light theme" : "Switch to dark theme"}
        {...props}
      >
        {resolvedTheme === "dark" ? (
          <IconSun size={iconSize[size]} stroke={1.5} />
        ) : (
          <IconMoon size={iconSize[size]} stroke={1.5} />
        )}
      </button>
    )
  }
  
  // Button variant
  if (variant === 'button') {
    return (
      <button
        type="button"
        onClick={toggleTheme}
        className={cn(
          "inline-flex items-center justify-center rounded-md border border-input bg-background px-4 py-2",
          textSizeClasses[size],
          "hover:bg-accent hover:text-accent-foreground",
          className
        )}
        {...props}
      >
        {resolvedTheme === "dark" ? "Light Mode" : "Dark Mode"}
      </button>
    )
  }
  
  // Switch variant
  return (
    <button
      type="button"
      onClick={toggleTheme}
      className={cn(
        "inline-flex items-center justify-center rounded-md px-3 py-2 font-medium transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring",
        textSizeClasses[size],
        resolvedTheme === "dark" 
          ? "bg-accent text-accent-foreground hover:bg-accent/80" 
          : "bg-muted text-muted-foreground hover:bg-muted/80",
        className
      )}
      aria-label={resolvedTheme === "dark" ? "Switch to light theme" : "Switch to dark theme"}
      {...props}
    >
      {resolvedTheme === "dark" ? (
        <span className="flex items-center">
          <IconSun size={iconSize[size]} stroke={1.5} className="mr-2" />
          Light
        </span>
      ) : (
        <span className="flex items-center">
          <IconMoon size={iconSize[size]} stroke={1.5} className="mr-2" />
          Dark
        </span>
      )}
    </button>
  )
}

/**

/**
 * Supported color theme values
 */
export type ThemeColor = 'brand' | 'error' | 'warning' | 'success';

/**
 * Supported component variant types
 */
export type ComponentVariant = 'primary' | 'secondary' | 'tertiary';

/**
 * Map design system colors to shadcn variants
 * 
 * @param color - The design system color
 * @returns The corresponding shadcn variant
 */
export const mapColorToShadcnVariant = (color: ThemeColor) => {
  switch (color) {
    case 'brand': return 'default';
    case 'error': return 'destructive';
    case 'warning':
    case 'success':
      return 'secondary'; // Will need additional classes
    default: return 'default';
  }
};

/**
 * Map design system variants to shadcn variants
 * 
 * @param variant - The design system variant
 * @returns The corresponding shadcn variant
 */
export const mapVariantToShadcnVariant = (variant: ComponentVariant) => {
  switch (variant) {
    case 'primary': return 'default';
    case 'secondary': return 'outline';
    case 'tertiary': return 'ghost';
    default: return 'default';
  }
};

/**
 * Generate Tailwind classes for color/variant combinations
 * 
 * @param color - The design system color
 * @param variant - The design system variant
 * @returns String of Tailwind classes
 */
export const getColorVariantClasses = (
  color: ThemeColor, 
  variant: ComponentVariant
): string => {
  const colorVariantMap: Record<ComponentVariant, Record<ThemeColor, string>> = {
    primary: {
      brand: 'bg-brand-600 hover:bg-brand-700 text-white focus-visible:ring-brand-500',
      error: 'bg-error-600 hover:bg-error-700 text-white focus-visible:ring-error-500',
      warning: 'bg-warning-600 hover:bg-warning-700 text-white focus-visible:ring-warning-500',
      success: 'bg-success-600 hover:bg-success-700 text-white focus-visible:ring-success-500',
    },
    secondary: {
      brand: 'border-brand-300 text-brand-700 hover:bg-brand-50 focus-visible:ring-brand-500',
      error: 'border-error-300 text-error-700 hover:bg-error-50 focus-visible:ring-error-500',
      warning: 'border-warning-300 text-warning-700 hover:bg-warning-50 focus-visible:ring-warning-500',
      success: 'border-success-300 text-success-700 hover:bg-success-50 focus-visible:ring-success-500',
    },
    tertiary: {
      brand: 'text-brand-700 hover:bg-brand-50 focus-visible:ring-brand-500',
      error: 'text-error-700 hover:bg-error-50 focus-visible:ring-error-500',
      warning: 'text-warning-700 hover:bg-warning-50 focus-visible:ring-warning-500',
      success: 'text-success-700 hover:bg-success-50 focus-visible:ring-success-500',
    }
  };

  return colorVariantMap[variant]?.[color] || '';
};

/**
 * Determine if the current theme is dark
 * 
 * @param theme - The current theme
 * @param systemTheme - The system theme
 * @returns True if the theme is dark
 */
export const isDarkTheme = (theme: string | undefined, systemTheme: string | undefined): boolean => {
  if (theme === 'system') {
    return systemTheme === 'dark';
  }
  return theme === 'dark';
};

import { clsx, type ClassValue } from 'clsx';
import { twMerge } from 'tailwind-merge';

/**
 * Combines multiple class values into a single className string
 * with proper Tailwind CSS class merging.
 * 
 * @param inputs - Class values to merge
 * @returns Merged className string
 */
export function cn(...inputs: ClassValue[]) {
  return twMerge(clsx(inputs));
}

/**
 * Utility function to generate a unique ID.
 * Useful for ensuring unique IDs for form elements.
 * 
 * @param prefix - Optional prefix for the ID
 * @returns A unique ID string
 */
export function generateId(prefix = 'id'): string {
  const randomId = Math.random().toString(36).substring(2, 9);
  return `${prefix}-${randomId}`;
}

/**
 * Type guard to check if a value is not null or undefined
 * 
 * @param value - Value to check
 * @returns True if value is not null or undefined
 */
export function isNotNullOrUndefined<T>(value: T | null | undefined): value is T {
  return value !== null && value !== undefined;
}

/**
 * Returns only the props that are safe to spread onto a DOM element
 * by filtering out custom props
 * 
 * @param props - The props object
 * @param propNames - Names of custom props to filter out
 * @returns Object with only DOM-safe props
 */
export function filterDOMProps<T extends Record<string, any>, K extends keyof T>(
  props: T, 
  propNames: K[]
): Omit<T, K> {
  const result = { ...props };
  propNames.forEach(name => delete result[name]);
  return result;
}
